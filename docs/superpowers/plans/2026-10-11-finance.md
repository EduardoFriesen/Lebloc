# Finanzas: plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Una pantalla `/finanzas` con cinco pestañas (Resumen, Ventas, Afluencia, Profesores y Clientes), filtro de período y comparación con el período anterior, para que el administrador tome decisiones.

**Architecture:**
- Las reglas puras van en `src/domain/`: `periods.ts` (rangos, comparación y buckets) y `finance.ts` (qué es ingreso y qué es egreso).
- `main/repos/finance.ts` hace consultas de agregación por día. `main/services/finance.ts` arma un DTO por pestaña y lo expone por un canal IPC cada una.
- En el renderer, `FinancePage` tiene el filtro y las pestañas como sub-rutas, cada pestaña con su `useAsync`. Los gráficos usan Recharts con un wrapper común.

**Tech Stack:** Electron, better-sqlite3, React 19, Tailwind 4, Zod 4, Recharts 3.10.1 (nueva), Vitest y Playwright.

**Spec:** `docs/superpowers/specs/2026-10-11-finance-design.md`

## Global Constraints

- **Ingreso del local:** `payments.local_cents` + kiosco (`quantity × unit_price_cents`) + `counter_sales.price_cents`.
- **Egreso:** `expenses.amount_cents` + `work_shifts.cost_cents` + costo del kiosco (`quantity × unit_cost_cents`).
- `payments.teacher_cents` nunca es ingreso: se muestra aparte como "Cobrado para profesores".
- Todo excluye lo anulado. Los rangos son inclusivos. `consumed_at` y `recorded_at` se pasan a hora local con `'localtime'`.
- El rango máximo es de 3 años (1096 días), y `from ≤ to`.
- **Paleta de series** (validada con `dataviz/scripts/validate_palette.js`: banda, croma, CVD adyacente y piso normal en PASS; contraste en WARN, que se cubre con tablas y tooltips):
  - claro: `#b0531f`, `#c8923a`, `#5a63b8`, `#4f8a3e`;
  - oscuro: `#a84a1e`, `#b98c30`, `#7a83dc`, `#5b9849`.
  - El orden es fijo, slot por entidad: Planes, Kiosco, Pases del día, Pédulas. Primera compra y renovación usan los slots 1 y 2.
  - La línea de egresos va en `--color-ink`. No se reusan colores de estado.
- Un solo eje Y por gráfico.
- Cada gráfico tiene un `<details>` "Ver como tabla" con los mismos datos, y cada componente sus 4 estados.

## Review Focus

1. **Rangos que cruzan años, febrero y meses de 30 días:** el período anterior tiene que caer bien. Tests en la Task 1.
2. **Un período sin ningún movimiento** muestra el estado vacío, sin gráficos rotos ni `NaN %`. Tests en las Tasks 1 (`percentChange` con 0) y 3 (totales en 0).
3. **Un pago de cuota de un plan vendido antes del rango** cuenta como ingreso del período (lo cobrado), pero no como plan vendido. Test en las Tasks 3 y 4.
4. **Los horarios cerca de medianoche:** la afluencia usa la hora local, no UTC. Test en la Task 5.
5. **Un rango inválido** (`desde` > `hasta` o más de 3 años) desde la URL: el esquema lo rechaza y la UI vuelve a "Este mes". Tests en las Tasks 3 y 9.

import { addMonths, daysBetween, parseIsoDate } from './dates';

export type Granularity = 'day' | 'week' | 'month';

export interface DateRange {
  from: string;
  to: string;
}

export const PERIOD_PRESETS = ['this_month', 'last_month', 'last_3_months', 'last_12_months', 'this_year'] as const;
export type PeriodPreset = (typeof PERIOD_PRESETS)[number];

const DAILY_UP_TO = 31;
const WEEKLY_UP_TO = 186;

export function addDays(iso: string, days: number): string {
  const [year, month, day] = parseIsoDate(iso);
  return new Date(Date.UTC(year, month - 1, day + days)).toISOString().slice(0, 10);
}

const monthStart = (iso: string) => `${iso.slice(0, 7)}-01`;
const monthEnd = (iso: string) => addDays(addMonths(monthStart(iso), 1), -1);
const monthIndex = (iso: string) => {
  const [year, month] = parseIsoDate(iso);
  return year * 12 + month;
};

export function presetRange(preset: PeriodPreset, today: string): DateRange {
  const thisMonth = monthStart(today);
  switch (preset) {
    case 'this_month':
      return { from: thisMonth, to: today };
    case 'last_month': {
      const from = addMonths(thisMonth, -1);
      return { from, to: monthEnd(from) };
    }
    case 'last_3_months':
      return { from: addMonths(thisMonth, -2), to: today };
    case 'last_12_months':
      return { from: addMonths(thisMonth, -11), to: today };
    case 'this_year':
      return { from: `${today.slice(0, 4)}-01-01`, to: today };
  }
}

/**
 * What a range is compared with. From the 1st of a month: the same months right before, up to the same day
 * ("this month so far" meets "last month up to today's day"; a whole month meets the whole month before).
 * Otherwise the same number of days right before.
 */
export function previousRange(range: DateRange): DateRange {
  if (range.from.endsWith('-01')) {
    const months = monthIndex(range.to) - monthIndex(range.from) + 1;
    const to = addMonths(range.to, -months);
    return { from: addMonths(range.from, -months), to: range.to === monthEnd(range.to) ? monthEnd(to) : to };
  }
  const days = daysBetween(range.from, range.to) + 1;
  return { from: addDays(range.from, -days), to: addDays(range.from, -1) };
}

export function granularityFor(range: DateRange): Granularity {
  const days = daysBetween(range.from, range.to) + 1;
  if (days <= DAILY_UP_TO) return 'day';
  return days <= WEEKLY_UP_TO ? 'week' : 'month';
}

/** The day itself, the Monday of its week, or its 'YYYY-MM' month. */
export function bucketKey(date: string, granularity: Granularity): string {
  if (granularity === 'day') return date;
  if (granularity === 'month') return date.slice(0, 7);
  const [year, month, day] = parseIsoDate(date);
  const weekday = new Date(Date.UTC(year, month - 1, day)).getUTCDay();
  return addDays(date, -((weekday + 6) % 7));
}

export function bucketKeys(range: DateRange, granularity: Granularity): string[] {
  const keys: string[] = [];
  const last = bucketKey(range.to, granularity);
  let cursor = granularity === 'month' ? monthStart(range.from) : bucketKey(range.from, granularity);
  for (let key = bucketKey(cursor, granularity); key <= last; key = bucketKey(cursor, granularity)) {
    keys.push(key);
    cursor = granularity === 'month' ? addMonths(cursor, 1) : addDays(cursor, granularity === 'week' ? 7 : 1);
  }
  return keys;
}

/** Adds daily rows into every bucket of the range, so empty buckets show as zero instead of a gap. */
export function bucketize<F extends string>(
  range: DateRange,
  granularity: Granularity,
  rows: readonly ({ date: string } & Record<F, number>)[],
  fields: readonly F[],
): ({ bucket: string } & Record<F, number>)[] {
  const zero = () => Object.fromEntries(fields.map((field) => [field, 0])) as Record<F, number>;
  const sums = new Map(bucketKeys(range, granularity).map((key) => [key, zero()]));
  for (const row of rows) {
    const sum = sums.get(bucketKey(row.date, granularity));
    if (sum) for (const field of fields) sum[field] += row[field];
  }
  return [...sums].map(([bucket, sum]) => ({ bucket, ...sum }));
}

/** Rounded change in percent; null when the previous value is 0 (no base to compare with). */
export function percentChange(current: number, previous: number): number | null {
  if (previous === 0) return null;
  return Math.round(((current - previous) / Math.abs(previous)) * 100);
}

```ts
export type MoneySource = 'plans' | 'teacher' | 'kiosk' | 'kiosk_cost' | 'day_pass' | 'shoe_rental' | 'expenses' | 'staff';
export type SourceTotals = Record<MoneySource, number>;

export interface Balance {
  incomeCents: number;
  expensesCents: number;
  resultCents: number;
}

export function emptyTotals(): SourceTotals {
  return { plans: 0, teacher: 0, kiosk: 0, kiosk_cost: 0, day_pass: 0, shoe_rental: 0, expenses: 0, staff: 0 };
}

export function sumBySource(rows: readonly { source: MoneySource; cents: number }[]): SourceTotals {
  const totals = emptyTotals();
  for (const row of rows) totals[row.source] += row.cents;
  return totals;
}

/**
 * What the local keeps. The teacher share of plan payments is passed on to the teacher, so it is never income;
 * kiosk stock is not an expense when bought but its cost is, sale by sale.
 */
export function balanceOf(totals: SourceTotals): Balance {
  const incomeCents = totals.plans + totals.kiosk + totals.day_pass + totals.shoe_rental;
  const expensesCents = totals.expenses + totals.staff + totals.kiosk_cost;
  return { incomeCents, expensesCents, resultCents: incomeCents - expensesCents };
}

export function marginPercent(balance: Balance): number | null {
  return balance.incomeCents === 0 ? null : Math.round((balance.resultCents / balance.incomeCents) * 100);
}
```

import { bucketize, type DateRange, granularityFor, previousRange } from '../../domain/periods';
import type { FinanceSummary } from '../../shared/types';
import type { Context } from '../context';
import * as repo from '../repos/finance';

const OUTFLOW = new Set(['expenses', 'staff', 'kiosk_cost']);
const SERIES_FIELDS = ['plans', 'kiosk', 'day_pass', 'shoe_rental', 'outflow'] as const;

export function getFinanceSummary(ctx: Context, range: DateRange): FinanceSummary {
  const previous = previousRange(range);
  const granularity = granularityFor(range);
  const rows = repo.moneyByDay(ctx.db, range);
  const totals = sumBySource(rows);
  const daily = rows.map((row) => ({
    date: row.date,
    plans: row.source === 'plans' ? row.cents : 0,
    kiosk: row.source === 'kiosk' ? row.cents : 0,
    day_pass: row.source === 'day_pass' ? row.cents : 0,
    shoe_rental: row.source === 'shoe_rental' ? row.cents : 0,
    outflow: OUTFLOW.has(row.source) ? row.cents : 0,
  }));
  return {
    range,
    previous,
    granularity,
    current: balanceOf(totals),
    before: balanceOf(sumBySource(repo.moneyByDay(ctx.db, previous))),
    totals,
    expensesByCategory: repo.expensesByCategory(ctx.db, range),
    series: bucketize(range, granularity, daily, SERIES_FIELDS),
  };
}

export function getFinanceTraffic(ctx: Context, range: DateRange): FinanceTraffic {
  const granularity = granularityFor(range);
  const rows = repo.visits(ctx.db, range);
  const heatmap = Array.from({ length: 7 }, () => Array.from({ length: 24 }, () => 0));
  const byKind: FinanceTraffic['byKind'] = { free: 0, teacher: 0, day_pass: 0 };
  for (const row of rows) {
    const day = heatmap[row.weekday];
    if (day) day[row.hour] = (day[row.hour] ?? 0) + row.visits;
    byKind[row.kind] += row.visits;
  }

    granularity,
    total: rows.reduce((total, row) => total + row.visits, 0),
    byKind,
    heatmap,
    byWeekday: heatmap.map((hours) => hours.reduce((total, count) => total + count, 0)),
    series: bucketize(range, granularity, rows, ['visits']).map(({ bucket, visits }) => ({ bucket, visits })),
  };

export function getFinanceClients(ctx: Context, range: DateRange): FinanceClients {
  const previous = previousRange(range);
  const granularity = granularityFor(range);
  const sales = repo.purchases(ctx.db, range).map((sale) => ({ date: sale.date, first: sale.renewal ? 0 : 1, renewal: sale.renewal ? 1 : 0 }));
  const { debtorCount, debtTotalCents } = summaryCounts(ctx.db);
  return {
    granularity,
    previous,
    newClients: repo.newClientCount(ctx.db, range),
    visitors: repo.visitorCount(ctx.db, range),
    firstPurchases: sales.reduce((total, sale) => total + sale.first, 0),
    renewals: sales.reduce((total, sale) => total + sale.renewal, 0),
    series: bucketize(range, granularity, sales, ['first', 'renewal']),
    lapsed: repo.lapsedClients(ctx.db, range, previous),
    debtorCount,
    debtTotalCents,
  };
}
